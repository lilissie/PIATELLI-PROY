  p.nom_perfil,
  p.rol_perfil,
  u.created_at as user_created
FROM profiles p
JOIN auth.users u ON p.id = u.id
WHERE p.email_perfil IN ('admin@piattelli.com');

-- Eliminar todas las tablas existentes (en orden correcto por dependencias)
DROP TABLE IF EXISTS HISTORIAL_TAREAS CASCADE;
DROP TABLE IF EXISTS TAREAS CASCADE;
DROP TABLE IF EXISTS RESERVAS_ACTIVIDADES CASCADE;
DROP TABLE IF EXISTS RESERVAS_HABITACIONES CASCADE;
DROP TABLE IF EXISTS VALORACIONES CASCADE;
DROP TABLE IF EXISTS RESERVAS CASCADE;
DROP TABLE IF EXISTS VENTAS_VINO CASCADE;
DROP TABLE IF EXISTS ACTIVIDADES CASCADE;
DROP TABLE IF EXISTS HABITACIONES CASCADE;
DROP TABLE IF EXISTS CLIENTES CASCADE;
DROP TABLE IF EXISTS VINOS CASCADE;


-- BODEGA PIATELLI -- CREACION DE TABLAS --
CREATE TABLE profiles (
  trabajador_id UUID NOT NULL,
  nom_perfil VARCHAR(100),
  rol_perfil VARCHAR(50),
  email_perfil VARCHAR(150),
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE CLIENTES (
  cliente_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nom_cli VARCHAR(100),
  apel_cli VARCHAR(100),
  dni_cli VARCHAR(20),
  email_cli VARCHAR(150),
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE VINOS (
  vino_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nombre_vino VARCHAR(100),
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE HABITACIONES (
  habitacion_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  numero_habitacion VARCHAR(10),
  tipo_habitacion VARCHAR(50),
  valoracion_prom DECIMAL(3,2),
  estado VARCHAR(30),
  capacidad_max INTEGER,
  precio_por_dia DECIMAL(10,2),
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE ACTIVIDADES (
  actividad_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  nombre_actividad VARCHAR(100),
  tipo VARCHAR(50),
  precio_por_dia DECIMAL(10,2),
  created_at TIMESTAMP DEFAULT NOW()
);

CREATE TABLE RESERVAS (
  reserva_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  cliente_id UUID NOT NULL,
  cantidad_personas INTEGER,
  fecha_inicio DATE,
  fecha_fin DATE,
  total_pagar DECIMAL(10,2),
  metodo_pago VARCHAR(50),
  estado VARCHAR(20),
  created_at TIMESTAMP DEFAULT NOW(),
  FOREIGN KEY (cliente_id) REFERENCES CLIENTES(cliente_id) ON DELETE CASCADE
);

CREATE TABLE VENTAS_VINO (
  venta_vino_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  fecha_venta_vino TIMESTAMP DEFAULT NOW(),
  cantidad_venta INTEGER,
  vino_id UUID NOT NULL,
  created_at TIMESTAMP DEFAULT NOW(),
  FOREIGN KEY (vino_id) REFERENCES VINOS(vino_id) ON DELETE CASCADE
);

CREATE TABLE VALORACIONES (
  valoracion_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  valoracion INTEGER,
  comentario TEXT,
  cliente_id UUID NOT NULL,
  habitacion_id UUID NOT NULL,
  created_at TIMESTAMP DEFAULT NOW(),
  FOREIGN KEY (cliente_id) REFERENCES CLIENTES(cliente_id) ON DELETE CASCADE,
  FOREIGN KEY (habitacion_id) REFERENCES HABITACIONES(habitacion_id) ON DELETE CASCADE
);

-- TABLAS INTERMEDIAS --

CREATE TABLE RESERVAS_HABITACIONES (
  reserva_id UUID NOT NULL,
  habitacion_id UUID NOT NULL,
  created_at TIMESTAMP DEFAULT NOW(),
  PRIMARY KEY (reserva_id, habitacion_id),
  FOREIGN KEY (reserva_id) REFERENCES RESERVAS(reserva_id) ON DELETE CASCADE,
  FOREIGN KEY (habitacion_id) REFERENCES HABITACIONES(habitacion_id) ON DELETE CASCADE
);

CREATE TABLE RESERVAS_ACTIVIDADES (
  reserva_id UUID NOT NULL,
  actividad_id UUID NOT NULL,
  fecha_inicio_act DATE,
  fecha_fin_act DATE,
  created_at TIMESTAMP DEFAULT NOW(),
  PRIMARY KEY (reserva_id, actividad_id),
  FOREIGN KEY (reserva_id) REFERENCES RESERVAS(reserva_id) ON DELETE CASCADE,
  FOREIGN KEY (actividad_id) REFERENCES ACTIVIDADES(actividad_id) ON DELETE CASCADE
);

CREATE TABLE TAREAS (
  tarea_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  prioridad INTEGER,
  nombre_tarea VARCHAR(100),
  fecha_tarea DATE,
  estado VARCHAR(20),
  trabajador_id UUID NOT NULL,
  created_at TIMESTAMP DEFAULT NOW(),
  FOREIGN KEY (trabajador_id) REFERENCES profiles(trabajador_id) ON DELETE CASCADE
);

CREATE TABLE HISTORIAL_TAREAS (
  historial_id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
  tarea_id UUID NOT NULL,
  fecha_baja DATE,
  created_at TIMESTAMP DEFAULT NOW(),
  FOREIGN KEY (tarea_id) REFERENCES TAREAS(tarea_id) ON DELETE CASCADE
);
