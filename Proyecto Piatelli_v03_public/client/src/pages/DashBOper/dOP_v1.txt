import { Info } from "@mui/icons-material";
import {
  Box,
  Card,
  CardContent,
  IconButton,
  Tooltip,
  Typography,
} from "@mui/material";
import { AnimatePresence, motion } from "framer-motion";
import { useState } from "react";
import { FiEdit } from "react-icons/fi";

import "./dashoperario.css";

const DashOp = () => {
  const [operario] = useState("María González");
  const [tipoSeleccionado, setTipoSeleccionado] = useState("");
  const [numeroSeleccionado, setNumeroSeleccionado] = useState("");
  const [habitacionSeleccionada, setHabitacionSeleccionada] = useState(null);

  const [dialogoAbierto, setDialogoAbierto] = useState(false);
  const [tareaEditando, setTareaEditando] = useState(null);
  const [paginaActual, setPaginaActual] = useState(1);

  const [habitaciones, setHabitaciones] = useState([
    { id: 1, tipo: "ejecutiva", numero: "101", estado: "libre" },
    { id: 2, tipo: "ejecutiva", numero: "102", estado: "libre" },
    { id: 3, tipo: "matrimonial", numero: "201", estado: "libre" },
    { id: 4, tipo: "matrimonial", numero: "202", estado: "libre" },
    { id: 5, tipo: "glamping", numero: "301", estado: "libre" },
    { id: 6, tipo: "glamping", numero: "302", estado: "libre" },
  ]);

  const [tareas, setTareas] = useState([
    {
      id: 1,
      nombre: "Limpiar habitación 101",
      fecha: "2024-01-15",
      estado: "pendiente",
      prioridad: "alta",
    },
    {
      id: 2,
      nombre: "Revisar sistema de aire",
      fecha: "2024-01-16",
      estado: "hecho",
      prioridad: "media",
    },
    {
      id: 3,
      nombre: "Reponer amenities",
      fecha: "2024-01-14",
      estado: "pendiente",
      prioridad: "baja",
    },
  ]);

  const tareasPorPagina = 8;

  // Ordenar tareas: primero pendientes, después hechas
  const tareasOrdenadas = [...tareas].sort((a, b) => {
    const prioridadOrden = { alta: 3, media: 2, baja: 1 };
    if (a.estado === "hecho" && b.estado !== "hecho") return 1;
    if (b.estado === "hecho" && a.estado !== "hecho") return -1;
    if (prioridadOrden[b.prioridad] !== prioridadOrden[a.prioridad]) {
      return prioridadOrden[b.prioridad] - prioridadOrden[a.prioridad];
    }
    return new Date(a.fecha) - new Date(b.fecha);
  });

  // Paginación
  const indiceUltimaTarea = paginaActual * tareasPorPagina;
  const indicePrimeraTarea = indiceUltimaTarea - tareasPorPagina;
  const tareasActuales = tareasOrdenadas.slice(
    indicePrimeraTarea,
    indiceUltimaTarea
  );
  const totalPaginas = Math.ceil(tareasOrdenadas.length / tareasPorPagina);

  const editarTarea = (tareaActualizada) => {
    setTareas(
      tareas.map((t) => (t.id === tareaActualizada.id ? tareaActualizada : t))
    );
  };

  const abrirDialogo = (tarea = null) => {
    setTareaEditando(tarea);
    setDialogoAbierto(true);
  };

  const cerrarDialogo = () => {
    setDialogoAbierto(false);
    setTareaEditando(null);
  };

  // 🔥 Ajustado para actualizar también la seleccionada
  const cambiarEstadoHabitacion = (id, nuevoEstado) => {
    const nuevasHabitaciones = habitaciones.map((hab) =>
      hab.id === id ? { ...hab, estado: nuevoEstado } : hab
    );
    setHabitaciones(nuevasHabitaciones);

    if (habitacionSeleccionada && habitacionSeleccionada.id === id) {
      setHabitacionSeleccionada({
        ...habitacionSeleccionada,
        estado: nuevoEstado,
      });
    }
  };

  const obtenerColorEstadoHabitacion = (estado) => {
    return estado === "libre" ? "#4CAF50" : "#FF9800";
  };

  const obtenerTextoEstadoHabitacion = (estado) => {
    return estado === "libre" ? "Libre" : "En Mantenimiento";
  };

  // Modal de edición de tarea
  const FormularioTarea = ({ tarea, onGuardar, onCancelar }) => {
    const [estado, setEstado] = useState(tarea.estado);

    const manejarGuardar = () => {
      onGuardar({ ...tarea, estado });
      onCancelar();
    };

    return (
      <div className="dialogo-overlay">
        <motion.div
          className="dialogo-contenido"
          initial={{ opacity: 0, scale: 0.8 }}
          animate={{ opacity: 1, scale: 1 }}
          exit={{ opacity: 0, scale: 0.8 }}
        >
          <h2 className="dialogo-titulo">Actualizar Estado</h2>
          <div className="form-group">
            <label className="form-label">Estado</label>
            <select
              className="form-select"
              value={estado}
              onChange={(e) => setEstado(e.target.value)}
            >
              <option value="pendiente">Pendiente</option>
              <option value="hecho">Hecho</option>
            </select>
          </div>
          <div className="dialogo-acciones">
            <button className="btn-cancelar" onClick={onCancelar}>
              Cancelar
            </button>
            <button className="btn-guardar" onClick={manejarGuardar}>
              Actualizar
            </button>
          </div>
        </motion.div>
      </div>
    );
  };

  return (
    <div className="dashboard-container">
      {/* Header */}
      <motion.div
        className="dashboard-header"
        initial={{ opacity: 0, y: -20 }}
        animate={{ opacity: 1, y: 0 }}
        transition={{ duration: 0.6 }}
      >
        <h1 className="dashboard-title">Dashboard - {operario}</h1>
      </motion.div>

      <motion.div className="dashboard-grid">
        {/* Sección de tareas */}
        <div className="tareas-section">
          <h2 className="tareas-title">Gestión de Tareas</h2>
          <div className="tabla-container">
            <table className="tabla-tareas">
              <thead>
                <tr>
                  <th>Prioridad</th>
                  <th>Tarea</th>
                  <th>Fecha</th>
                  <th>Estado</th>
                  <th>Acciones</th>
                </tr>
              </thead>
              <tbody>
                <AnimatePresence>
                  {tareasActuales.map((tarea) => (
                    <motion.tr
                      key={tarea.id}
                      initial={{ opacity: 0, height: 0 }}
                      animate={{ opacity: 1, height: "auto" }}
                      exit={{ opacity: 0, height: 0 }}
                      transition={{ duration: 0.3 }}
                      className={
                        tarea.estado === "hecho" ? "tarea-completada" : ""
                      }
                    >
                      <td>
                        <div
                          className={`prioridad-indicator prioridad-${tarea.prioridad}`}
                        />
                      </td>
                      <td>{tarea.nombre}</td>
                      <td>{new Date(tarea.fecha).toLocaleDateString()}</td>
                      <td>
                        <span className={`estado-chip estado-${tarea.estado}`}>
                          {tarea.estado}
                        </span>
                      </td>
                      <td>
                        <button
                          className="btn-accion btn-editar"
                          onClick={() => abrirDialogo(tarea)}
                        >
                          <FiEdit size={18} />
                        </button>
                      </td>
                    </motion.tr>
                  ))}
                </AnimatePresence>
              </tbody>
            </table>

            {/* Paginación */}
            {totalPaginas > 1 && (
              <div className="paginacion">
                {Array.from({ length: totalPaginas }, (_, i) => i + 1).map(
                  (pagina) => (
                    <button
                      key={pagina}
                      className={`btn-pagina ${
                        pagina === paginaActual ? "active" : ""
                      }`}
                      onClick={() => setPaginaActual(pagina)}
                    >
                      {pagina}
                    </button>
                  )
                )}
              </div>
            )}
          </div>
        </div>

        {/* Sección de mantenimiento */}
        <motion.div className="metrica-card">
          <Box sx={{ mb: 2, display: "flex", justifyContent: "space-between" }}>
            <Typography variant="h6">Mantenimiento de Habitaciones</Typography>
            <Tooltip title="Gestionar estado de las habitaciones">
              <IconButton size="small">
                <Info sx={{ fontSize: 18, color: "#666" }} />
              </IconButton>
            </Tooltip>
          </Box>

          {/* Selects dependientes */}
          <Box sx={{ display: "flex", gap: 2, mb: 2 }}>
            <select
              className="form-select"
              value={tipoSeleccionado || ""}
              onChange={(e) => {
                setTipoSeleccionado(e.target.value);
                setNumeroSeleccionado("");
                setHabitacionSeleccionada(null);
              }}
            >
              <option value="">Seleccionar tipo</option>
              {[...new Set(habitaciones.map((h) => h.tipo))].map((tipo) => (
                <option key={tipo} value={tipo}>
                  {tipo}
                </option>
              ))}
            </select>

            {tipoSeleccionado && (
              <select
                className="form-select"
                value={numeroSeleccionado || ""}
                onChange={(e) => {
                  setNumeroSeleccionado(e.target.value);
                  const hab = habitaciones.find(
                    (h) =>
                      h.tipo === tipoSeleccionado && h.numero === e.target.value
                  );
                  setHabitacionSeleccionada(hab);
                }}
              >
                <option value="">Seleccionar número</option>
                {habitaciones
                  .filter((h) => h.tipo === tipoSeleccionado)
                  .map((h) => (
                    <option key={h.id} value={h.numero}>
                      {h.numero}
                    </option>
                  ))}
              </select>
            )}
          </Box>

          {/* Mostrar habitación seleccionada */}
          {habitacionSeleccionada && (
            <Card sx={{ mt: 2 }}>
              <CardContent
                sx={{
                  p: 2,
                  display: "flex",
                  flexDirection: "column",
                  gap: 2,
                }}
              >
                <Typography variant="subtitle1">
                  {habitacionSeleccionada.tipo} -{" "}
                  {habitacionSeleccionada.numero}
                </Typography>
                <Typography
                  variant="body2"
                  sx={{
                    color: obtenerColorEstadoHabitacion(
                      habitacionSeleccionada.estado
                    ),
                  }}
                >
                  {obtenerTextoEstadoHabitacion(habitacionSeleccionada.estado)}
                </Typography>

                {/* Botón para cambiar estado */}
                <button
                  className="btn-estado"
                  onClick={() =>
                    cambiarEstadoHabitacion(
                      habitacionSeleccionada.id,
                      habitacionSeleccionada.estado === "libre"
                        ? "mantenimiento"
                        : "libre"
                    )
                  }
                >
                  Cambiar a{" "}
                  {habitacionSeleccionada.estado === "libre"
                    ? "Mantenimiento"
                    : "Libre"}
                </button>

                {/* Botón para liberar selección */}
                <button
                  className="btn-liberar"
                  onClick={() => {
                    setTipoSeleccionado("");
                    setNumeroSeleccionado("");
                    setHabitacionSeleccionada(null);
                  }}
                >
                  Liberar Selección
                </button>
              </CardContent>
            </Card>
          )}
        </motion.div>
      </motion.div>

      {/* Diálogo de edición */}
      <AnimatePresence>
        {dialogoAbierto && (
          <FormularioTarea
            tarea={tareaEditando}
            onGuardar={editarTarea}
            onCancelar={cerrarDialogo}
          />
        )}
      </AnimatePresence>
    </div>
  );
};

export default DashOp;
